3. Google 後台試算表備份與本機資料管理

設定 Google Apps Script Web App 網址後，可一鍵將全公司專案與請款資料同步至 Google 試算表，或進行本機 JSON 備份匯出與匯入。

請將以下程式碼貼入 Google 試算表 > 擴充功能 > Apps Script 中並部署為網頁應用程式： 
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('AHEAD_DB') || ss.insertSheet('AHEAD_DB');
  var raw = sheet.getRange('A1').getValue();
  return ContentService.createTextOutput(raw || '{"projects":[]}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('AHEAD_DB') || ss.insertSheet('AHEAD_DB');
  var payload = e.postData.contents;
  sheet.getRange('A1').setValue(payload);
  sheet.getRange('A2').setValue('最後備份時間：' + new Date().toLocaleString('zh-TW'));
  return ContentService.createTextOutput(JSON.stringify({ ok: true }))
    .setMimeType(ContentService.MimeType.JSON);
}
